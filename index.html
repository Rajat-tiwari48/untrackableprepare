1 🔥 Rotting Oranges — BFS multi-source, return min minutes LeetCode 994Glassdoor
▼

Problem: Grid with 0=empty,1=fresh,2=rotten. Each minute rotten spreads to 4-adjacent fresh. Return min minutes for all to rot, or -1.

function orangesRotting(grid) {
  const rows=grid.length, cols=grid[0].length;
  let fresh=0;
  const q=[];
  for(let r=0;r<rows;r++) for(let c=0;c<cols;c++) {
    if(grid[r][c]===1) fresh++;
    if(grid[r][c]===2) q.push([r,c,0]);
  }
  if(fresh===0) return 0;
  const dirs=[[1,0],[-1,0],[0,1],[0,-1]];
  let maxTime=0;
  for(let i=0;i<q.length;i++) {
    const[r,c,t]=q[i];
    for(const[dr,dc] of dirs) {
      const nr=r+dr, nc=c+dc;
      if(nr>=0&&nr<rows&&nc>=0&&nc<cols&&grid[nr][nc]===1) {
        grid[nr][nc]=2; fresh--; maxTime=Math.max(maxTime,t+1); q.push([nr,nc,t+1]);
      }
    }
  }
  return fresh===0?maxTime:-1;
} // TC: O(m*n)
2 🔥 LRU Cache — implement using HashMap + Doubly Linked List LeetCode 146Glassdoor/LPU
▼

Problem: Design LRU Cache with O(1) get and put.

class LRUCache {
  constructor(cap) {
    this.cap=cap; this.map=new Map();
    this.head={key:0,val:0,prev:null,next:null};
    this.tail={key:0,val:0,prev:null,next:null};
    this.head.next=this.tail; this.tail.prev=this.head;
  }
  _remove(node){node.prev.next=node.next;node.next.prev=node.prev;}
  _insert(node){node.prev=this.tail.prev;node.next=this.tail;this.tail.prev.next=node;this.tail.prev=node;}
  get(key){
    if(!this.map.has(key)) return -1;
    const node=this.map.get(key);
    this._remove(node);this._insert(node);
    return node.val;
  }
  put(key,val){
    if(this.map.has(key)) this._remove(this.map.get(key));
    const node={key,val,prev:null,next:null};
    this._insert(node);this.map.set(key,node);
    if(this.map.size>this.cap){const lru=this.head.next;this._remove(lru);this.map.delete(lru.key);}
  }
}
3 🔥 Interleaving String — 2D DP (Hard) LeetCode 97LeetCode Sept 2025

Problem: Check if s3 is formed by interleaving s1 and s2.

function isInterleave(s1,s2,s3){
  const m=s1.length,n=s2.length;
  if(m+n!==s3.length) return false;
  const dp=Array.from({length:m+1},()=>new Array(n+1).fill(false));
  dp[0][0]=true;
  for(let i=1;i<=m;i++) dp[i][0]=dp[i-1][0]&&s1[i-1]===s3[i-1];
  for(let j=1;j<=n;j++) dp[0][j]=dp[0][j-1]&&s2[j-1]===s3[j-1];
  for(let i=1;i<=m;i++) for(let j=1;j<=n;j++)
    dp[i][j]=(dp[i-1][j]&&s1[i-1]===s3[i+j-1])||(dp[i][j-1]&&s2[j-1]===s3[i+j-1]);
  return dp[m][n];
}
4 🔥 Maximum Product of Word Lengths — Bitmask LeetCode 318LeetCode Sept 2025

Problem: Find max length(word[i]) * length(word[j]) where the two words share no common letters.

function maxProduct(words){
  const masks=words.map(w=>w.split('').reduce((m,c)=>m|(1<<(c.charCodeAt(0)-97)),0));
  let max=0;
  for(let i=0;i<words.length;i++)
    for(let j=i+1;j<words.length;j++)
      if((masks[i]&masks[j])===0) max=Math.max(max,words[i].length*words[j].length);
  return max;
}

function decodeString(s){
  let stack=[],curr='',num=0;
  for(let c of s){
    if(!isNaN(c)) num=num*10+Number(c);
    else if(c==='['){stack.push([curr,num]);curr='';num=0;}

9 🔥 Quick Sort — full implementation, explain complexity 2020 TR1
▼ 
function qsort(a,lo=0,hi=a.length-1){
  if(lo<hi){let p=part(a,lo,hi);qsort(a,lo,p-1);qsort(a,p+1,hi);}
  return a;
}
function part(a,lo,hi){
  let piv=a[hi],i=lo-1;
  for(let j=lo;j<hi;j++)if(a[j]<=piv){i++;[a[i],a[j]]=[a[j],a[i]];}
  [a[i+1],a[hi]]=[a[hi],a[i+1]];
  return i+1;
} // avg O(nlogn) | worst O(n²)
10 🔥 Combination Sum — backtracking 2022 Coding Round

function combinationSum(candidates,target){
  const res=[];
  function bt(start,curr,rem){
    if(rem===0){res.push([...curr]);return;}
    for(let i=start;i<candidates.length;i++){
      if(candidates[i]<=rem) bt(i,curr.concat(candidates[i]),rem-candidates[i]);
    }
  }
  bt(0,[],target);
  return res;
}

21 Minimum number of swaps to sort array Naukri 2019

function minSwaps(arr){
  const sorted=[...arr].sort((a,b)=>a-b);
  const pos=new Map(arr.map((v,i)=>[v,i]));
  let swaps=0;
  for(let i=0;i<arr.length;i++){
    if(arr[i]!==sorted[i]){
      swaps++;
      const j=pos.get(sorted[i]);
      pos.set(arr[i],j);
      [arr[i],arr[j]]=[arr[j],arr[i]];
    }
  }
  return swaps;
}
